#
#   שורה שמתחילה ב-# היא הערה. שמות גיליונות וערכי טקסט - בתוך "מירכאות".
#   עמודות כותבים באותיות: A, T, AA ...
#   BASE  = גיליון הבסיס (הקובץ הגדול שמתעדכן)
#   UPDATE = גיליון העדכון (הנתונים שנכנסו מהקובץ החיצוני)

# ---------------- שלב 1: ייבוא קבצים ----------------
# IMPORT <מספר> "<שם הגיליון בקובץ הגדול>"   [FROM "<שם הגיליון בקובץ הנכנס>"]
# הגיליון בקובץ הגדול נמחק ומוחלף כולו בגיליון מהקובץ הנכנס (באותו מיקום).
# בלי FROM - נלקח הגיליון הראשון של הקובץ הנכנס.
# אם הגיליון לא קיים בקובץ הגדול - הוא ייווצר בסוף.

IMPORT 1 "INTEGRATOR"
IMPORT 2 "SHEET2"            # <- שנה לשם הגיליון האמיתי של הקובץ השני


# ---------------- שלב 2: האלגוריתם ----------------
ALGORITHM "עדכון סטטוס CDF מ-INTEGRATOR"

BASE FIRST                   # FIRST = הגיליון הראשון בחוברת. אפשר גם: BASE "שם הגיליון"
UPDATE "INTEGRATOR"

BASE_START_ROW 3             # שורת הנתונים הראשונה בגיליון הבסיס (מתחת לכותרות)
UPDATE_START_ROW 2           # שורת הנתונים הראשונה בגיליון העדכון

# השוואה: עמודה A בעדכון מול עמודה C בבסיס (בלי רגישות לרווחים ולאותיות גדולות/קטנות)
MATCH UPDATE.A = BASE.C

# האם לדלג על תא ריק בקובץ העדכון (YES = לא לדרוס ערך קיים בריק)
SKIP_EMPTY NO

DATE_FORMAT "dd/mm/yyyy"

# 1. העתקת הסטטוס
COPY UPDATE.M -> BASE.T

# 2+3. אם הסטטוס החדש הוא FINAL APPROVAL - העתקת השם ועמודות Q,R
IF BASE.T = "9 - FINAL APPROVAL" COPY UPDATE.N -> BASE.X
IF BASE.T = "9 - FINAL APPROVAL" COPY UPDATE.Q -> BASE.AA
IF BASE.T = "9 - FINAL APPROVAL" COPY UPDATE.R -> BASE.AB

# 4. אם משהו בשורה השתנה בפועל - תאריך היום בעמודה R
ON_CHANGE SET BASE.R = TODAY


# ---------------- שורה חדשה לחריג (מזוג PDF) ----------------
# Claude ממלא את השדות מתוך PDF החריג ו-PDF ה-ATT (דרך claude.ai, העתק-הדבק),
# ואתה עובר על הטופס ומאשר. אם ה-CDF כבר קיים בעמודת המפתח - השורה שלו מתעדכנת,
# אחרת נוספת שורה חדשה אחרי השורה האחרונה.
#
# מקורות אפשריים לכל עמודה:
#   CDF            מספר ה-CDF (משם הקובץ, נבדק מול הטופס)
#   TODAY / NOW    תאריך היום / תאריך ושעה
#   "טקסט"         ערך קבוע
#   AI.<שדה>       DATE  PRIORITY  DESCRIPTION  OEM  DRAWING  INSTALLATION  CODE  INITIATOR
#   LINK MAIN "<תיקייה>"   קישור ל-PDF החריג (התיקייה + שם הקובץ)
#   LINK ATT  "<תיקייה>"   קישור ל-PDF ה-ATT
#
# !!! לאמת את אותיות העמודות מול שורת הכותרות בקובץ !!!
# הסדר כאן לפי מבנה השורות מ-26/10. החלונית מציגה ליד כל שדה את שם הכותרת
# של העמודה שהוא נכתב אליה - אם זה לא מתאים, תקן כאן.

NEWROW_SHEET FIRST
NEWROW_KEY C
NEWROW_HEADER_ROW 1

NEWROW A = LINK ATT ""        # <- שים בין המירכאות את התיקייה של קבצי ה-PDF
NEWROW B = LINK MAIN ""
NEWROW C = CDF
NEWROW F = TODAY
NEWROW G = AI.PRIORITY
NEWROW H = AI.DESCRIPTION
NEWROW I = AI.OEM
NEWROW J = AI.DRAWING
NEWROW K = AI.INSTALLATION
NEWROW L = AI.CODE
NEWROW M = "-"
NEWROW O = AI.INITIATOR

# ההנחיה ל-Claude נמצאת בקובץ ai-prompt.txt - אפשר לערוך.


# ---------------- מייל ----------------
# דוחות שאפשר לשלוח במייל, בנוסף ל"שינויים מההרצה האחרונה" ו"שורות חדשות היום".
# REPORT "<נושא המייל>" WHERE <תנאי> [AND <תנאי>] COLUMNS <עמודות>
REPORT "CDFs in FINAL APPROVAL" WHERE BASE.T = "9 - FINAL APPROVAL" COLUMNS C, T, X
REPORT "Open CDFs - CONV. ENG." WHERE BASE.T = "3 - CONV. ENG." COLUMNS C, T, R

# קבוצות נמענים. שים לב: אם המאגר ב-GitHub ציבורי, כל אחד יכול לקרוא את הקובץ הזה -
# עדיף לשמור קבוצות בחלונית עצמה (נשמרות רק במחשב שלך).
# EMAIL_GROUP "צוות" "name1@company.com; name2@company.com"


# ---------------- יומן שינויים ----------------
# כל הרצה מוסיפה לגיליון הזה את כל השינויים (ערך ישן -> חדש) ו-CDF שלא נמצאו.
# LOG NONE = בלי יומן.
LOG "UPDATE LOG"
